```VHDL
library IEEE;
use IEEE.std_logic_vector6411.all;
use IEEE.numeric_std.all;

entity timer is
port(
	clk : in STD_LOGIC;
	enable : in STD_LOGIC;
	load : in STD_LOGIC;
	M : in STD_LOGIC_VECTOR(31 downto 0);
	Q : out STD_LOGIC
);
end timer;
```

## 5.
```VHDL
architecture rtl of timer is

signal M32 : STD_LOGIC_VECTOR(31 downto 0);
signal R32 : STD_LOGIC_VECTOR(31 downto 0);
signal Mux32 : STD_LOGIC_VECTOR(31 downto 0);
signal C : UNSIGNED;

signal OR1 : STD_LOGIC;

begin

-- Processus pour le registre
process(clk)
begin
	if clk'event and clk='1' then
		if load='1' then
			R32<=M32;
		end if;
	end if;
end process;


-- Processus pour le décompteur
process(clk)
begin
	if clk'event and clk='1' then
		if enable='1' then -- On choisit enable prioritaire
			if OR1='1' then
				C<=unsigned(M32);
			else
				if C=0 then
					tc<='0';
					C<=unsigned(M32);
				else if C=1 then
					tc<='1';
					C<=0;
				else
					tc<='0';
					C <= C-1;
				end if;
		end if;
	end if;
end process;
end rtl;
```

```VHDL
-- Instruction concurrente pour M32
M32<=STD_LOGIC_VECTOR(unsigned(M)-1);

-- Instruction concurrente pour Mux32
Mux32<=R32;

```
